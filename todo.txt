def print_tracemalloc_diff(snapshot1, snapshot2, title="Memory Difference"):
    """
    Compares two tracemalloc snapshots and prints the top differences.
    """
    print(f"\n--- {title} ---")
    top_stats = snapshot2.compare_to(snapshot1, 'lineno')

    print("Top 10 memory differences:")
    for stat in top_stats[:10]:
        print(stat)

    print("--------------------")

def print_tracemalloc_stats(snapshot, title="Final Memory Statistics"):
    """
    Prints overall memory statistics from a single snapshot.
    """
    print(f"\n--- {title} ---")
    top_stats = snapshot.statistics('lineno')

    print("Top 10 memory blocks:")
    for stat in top_stats[:10]:
        print(stat)

    total_memory_bytes = sum(stat.size for stat in top_stats)
    print(f"Total allocated memory: {total_memory_bytes / (1024*1024):.2f} MB")
    print(f"Total number of blocks: {len(top_stats)}")
    print("--------------------")
